# === Variables ===
varlist board 9
var is_running 1
var winner 0
var cell 0
var input_pos 0
var curr_pl 1
var flag 0
var idx 0
var loop_ctl 1

# === Main Game Loop ===
while is_running

    put loop_ctl 1
    put idx 0

    # Draw Board
    while loop_ctl
        getlist cell board idx
        
        if : cell = 0
            prints *|
        }
        if : cell = 1
            prints X|
        }
        if : cell = 2
            prints O|
        }

        if : idx = 2
            prints ^------^
        }
        if : idx = 5
            prints ^------^
        }
        if : idx = 8
            prints ^
        }
        
        cont idx idx + 1
        
        if : idx = 9
            put loop_ctl 0
        }
    }

    # User Input
    prints Enter the corresponding coordinates(1-9)
    inputi input_pos
    cont input_pos input_pos - 1
    setlist board input_pos curr_pl

    # Switch Player
    if : curr_pl = 1
        put curr_pl 2
    }
    else
        put curr_pl 1
    }

    # === Win Check Logic ===
    var p_check 1
    put loop_ctl 1

    while loop_ctl
        var v1 0
        var v2 0
        var v3 0
        var val 0
        var inner_loop 1
        put idx 0

        # Check Rows
        while inner_loop
            getlist val board idx
            is v1 val = p_check
            cont idx idx + 1
            
            getlist val board idx
            is v2 val = p_check
            cont idx idx + 1
            
            getlist val board idx
            is v3 val = p_check
            cont idx idx + 1

            is flag v1 and v2
            is flag flag and v3
            if flag
                copy p_check winner
                put is_running 0
                put inner_loop 0
            }
            
            if : idx = 9
                put inner_loop 0
            }
        }

        put inner_loop 1
        put idx 0

        # Check Columns
        while inner_loop
            getlist val board idx
            is v1 val = p_check
            cont idx idx + 3
            
            getlist val board idx
            is v2 val = p_check
            cont idx idx + 3
            
            getlist val board idx
            is v3 val = p_check
            cont idx idx + 3

            is flag v1 and v2
            is flag flag and v3
            if flag
                copy p_check winner
                put is_running 0
                put inner_loop 0
            }

            if : idx = 12
                put inner_loop 0
            }
            cont idx idx - 8
        }

        # Check Diagonals
        getlist val board 0
        is v1 val = p_check
        getlist val board 4
        is v2 val = p_check
        getlist val board 8
        is v3 val = p_check
        
        is flag v1 and v2
        is flag flag and v3
        if flag
            copy p_check winner
            put is_running 0
        }

        getlist val board 2
        is v1 val = p_check
        getlist val board 4
        is v2 val = p_check
        getlist val board 6
        is v3 val = p_check
        
        is flag v1 and v2
        is flag flag and v3
        if flag
            copy p_check winner
            put is_running 0
        }

        cont p_check p_check + 1

        if : p_check = 3
            put loop_ctl 0
        }
    }
}

# === Game Over ===
if : winner = 1
    prints X is win!
}
else
    prints O is win!
}
